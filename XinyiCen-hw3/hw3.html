Problem 1

  1. Each row represents one face image.
  2. Each column represents one pixel position.
  3. V encodes the orignial images: each of row of V contains the r weights describing one image as a mixture of the shared patterns stored in W.
  4. W represents the dictonary, since it stores the patterns that give meaning to the weights in V.
  5. Each image has 16384 (by \(128 \times 128\)) pixels. Then X matrix has memory of 6553600000 (by \(50000 \times 16384 \times 8\)) bytes. V matrix has memory of 400000000 (by \(50000 \times 1000 \times 8\)) bytes. W matrix has memory of 131072000 (by \(1000 \times 16384 \times 8\)) bytes. The NMF uses 531072000 (by \(400000000+131072000\)) bytes. The memory saving with NMF is 6022528000 (by \(6553600000-531072000\)) bytes. (around 91.9 percent is saved)

Problem 2

  • The updated V should be: \[ v_{ik}^{(t+1)} = v_{ik}^{(t)} \frac{\sum_j x_{ij}w_{kj}^{(t)}}{\sum_j b_{ij}^{(t)}w_{kj}^{(t)}} = V^{(t)} \odot \frac{X(W^{(t)})^\top}{(V^{(t)}W^{(t)})(W^{(t)})^\top} \]

  • The updated W should be: \[ w_{kj}^{(t+1)}=w_{kj}^{(t)}\frac{\sum_i x_{ij} v_{ik}^{(t+1)}}{\sum_i b_{ij}^{(t+1/2)} v_{ik}^{(t+1)}} = W^{(t)} \odot \frac{(V^{(t+1)})^\top X}{(V^{(t+1)})^\top (V^{(t+1)}W^{(t)})} \]

using Random, LinearAlgebra

function nnmf(
  # Positional arguments
  X       :: Matrix{T}, 
  r       :: Integer;
  # Keyword arguments
  maxiter :: Integer = 1000, 
  tolfun  :: Real = 1e-4,
  V       :: Matrix{T} = Random.rand!(similar(X, size(X, 1), r)),
  W       :: Matrix{T} = Random.rand!(similar(X, r, size(X, 2))),
  ) where T <: AbstractFloat

    ### Allocate storage once: use similar() create an empty workspace with same element type
    B = similar(X)      # Store the V * W
    num_V = similar(V)  # Store X * transpose W
    deno_V = similar(V) # Store (V*W) * transpose W
    num_W = similar(W)  # Store transpose V * X
    deno_W = similar(W) # Store transpose V * (V*W)
  
    ### L(V,W) Original object
    # put V*W into the exsiting matrix B
    mul!(B,V,W)  
    # acculate squared difference without creating a residual matrix
    obj = zero(T)
    # @inbounds skips bounds check, eachindex visits every entry
    @inbounds for i in eachindex(X,B)
        obj += abs2(X[i]-B[i])
    end


  ### Stop status and number of interations
  conv = false
  niter = 0

  for iter in 1:maxiter
    ### Record the original object
    obj_old = obj

    ### Update V suing current W and B
    # V .*= (X*transpose(W)) ./ ((V*W)*transpose(W))
    mul!(num_V, X, transpose(W))
    mul!(deno_V, B, transpose(W))
        @. V *= num_V/deno_V

    # Recompute V*W  using updated V
    mul!(B, V, W)

    ### Update W using updated V
    # W .*= (transpose(V)*X) ./ (transpose(V)*(V*W))
    mul!(num_W, transpose(V), X)
    mul!(deno_W, transpose(V), B)
    @. W *= num_W/deno_W

    ### Update objects with updated W and V
    mul!(B,V,W)
    obj = zero(T)   # reset object before calculate current error
    @inbounds for i in eachindex(X,B)
        obj += abs2(X[i]-B[i])
    end


    ### record every iteration
    niter = iter

    # convergence check
    if abs(obj - obj_old) < tolfun * (abs(obj_old) + 1)
        conv = true
        break
    end
  end

  # Output: Formatted as a NamedTuple
  result = (;
    V=V,
    W=W,
    r=r,
    converged=conv,    # TODO: did the algorithm converge? 
    objective=obj,  # TODO: what is the objective at the end?
    iterations=niter       # TODO: how many iterations were taken?
  )
  return result
end
